#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        bool have[26] = {};
        for (int i = 0; i < n; i++) {
            string s;
            cin >> s;
            have[s[0] - 'a'] = true;
        }
        vector<string> a(m);
        for (auto &s : a) cin >> s;
        vector<bool> done(m);
        while (true) {
            bool found = false;
            for (int i = 0; i < m; i++) {
                if (done[i]) continue;
                bool possible = true;
                for (char c : a[i])
                    if (!have[c - 'A']) { possible = false; break; }
                if (possible) {
                    done[i] = true;
                    have[a[i][0] - 'A'] = true;
                    found = true;
                }
            }
            if (!found) break;
        }
        bool ok = true;
        for (bool x : done)
            if (!x) { ok = false; break; }
        cout << (ok ? "YES" : "NO") << endl;
    }
}
